// ================================================= // GUARDAR NOTAS // ================================================= async function guardarNotas() { if (!cursoSeleccionado) { alert("Primero selecciona un curso."); return false; } if (actividades.length === 0) { alert("Este curso no tiene actividades."); return false; } const inputs = document.querySelectorAll(".nota-input"); // ============================================= // VALIDAR TODAS LAS NOTAS // ============================================= for (const input of inputs) { if (input.value === "") { continue; } const valor = Number(input.value); const maximo = Number(input.dataset.maximo); if (valor < 0 || valor > maximo) { alert( "⚠️ NO SE PUEDEN GUARDAR LAS NOTAS\n\n" + "La actividad \"" + obtenerNombreActividad( input.dataset.actividad ) + "\" tiene un máximo de " + maximo + " puntos.\n\n" + "La nota ingresada es: " + valor ); input.focus(); return false; } } mostrarMensaje( "💾 Guardando notas..." ); // ============================================= // PREPARAR REGISTROS // ============================================= const registros = []; inputs.forEach(input => { const estudianteId = Number( input.dataset.estudiante ); const actividadId = Number( input.dataset.actividad ); let nota = null; if (input.value !== "") { nota = Number(input.value); } registros.push({ actividad_id: actividadId, estudiante_id: estudianteId, nota: nota }); }); // ============================================= // GUARDAR CALIFICACIONES // ============================================= const { error: errorCalificaciones } = await supabaseClient .from("calificaciones") .upsert( registros, { onConflict: "actividad_id,estudiante_id" } ); if (errorCalificaciones) { console.error( errorCalificaciones ); mostrarMensaje( "❌ Error al guardar calificaciones: " + errorCalificaciones.message, true ); return false; } // ============================================= // CALCULAR Y GUARDAR NOTA DEL 3T // ============================================= const estudiantesCurso = estudiantes.filter( e => e.curso === cursoSeleccionado ); for ( const estudiante of estudiantesCurso ) { const totales = { SER: [], SABER: [], HACER: [], AUTOEVALUACION: [] }; actividades.forEach( actividad => { const input = document.querySelector( `.nota-input[data-estudiante="${estudiante.id}"][data-actividad="${actividad.id}"]` ); if ( !input || input.value === "" ) { return; } const nota = Number(input.value); if (!isNaN(nota)) { totales[ actividad.categoria ].push(nota); } }); // ========================================= // PROMEDIOS // ========================================= const promedioSER = calcularPromedio( totales.SER ); const promedioSABER = calcularPromedio( totales.SABER ); const promedioHACER = calcularPromedio( totales.HACER ); const promedioAUTO = calcularPromedio( totales.AUTOEVALUACION ); // ========================================= // CONVERTIR A PUNTAJE // ========================================= const puntosSER = (promedioSER / 100) * LIMITES.SER; const puntosSABER = (promedioSABER / 100) * LIMITES.SABER; const puntosHACER = (promedioHACER / 100) * LIMITES.HACER; const puntosAUTO = (promedioAUTO / 100) * LIMITES.AUTOEVALUACION; // ========================================= // NOTA FINAL DEL 3T // ========================================= const nota3T = Math.round( puntosSER + puntosSABER + puntosHACER + puntosAUTO ); // ========================================= // ACTUALIZAR SUPABASE // ========================================= const { error } = await supabaseClient .from("matematica") .update({ nota_3t: nota3T }) .eq( "id", estudiante.id ); if (error) { console.error(error); mostrarMensaje( "❌ Error al actualizar 3T: " + error.message, true ); return false; } } // ============================================= // RECARGAR DATOS // ============================================= await cargarEstudiantes(); await cargarActividades(); mostrarTablaGeneral(); mostrarMensaje( "✓ Notas guardadas correctamente." ); return true; } // ================================================= // EXPORTAR EXCEL // PRIMERO GUARDA LAS NOTAS // ================================================= async function exportarExcel() { try { // ========================================= // PASO 1: GUARDAR NOTAS // ========================================= if (!cursoSeleccionado) { alert( "Primero selecciona un curso." ); return; } if (actividades.length === 0) { alert( "Este curso no tiene actividades." ); return; } mostrarMensaje( "💾 Guardando notas antes de exportar..." ); const guardado = await guardarNotas(); // ========================================= // SI NO SE GUARDÓ, NO EXPORTAR // ========================================= if (!guardado) { mostrarMensaje( "❌ No se pudo guardar. " + "El Excel no fue generado.", true ); return; } // ========================================= // PASO 2: PREPARAR EXCEL // ========================================= mostrarMensaje( "📊 Preparando archivo Excel..." ); if ( typeof XLSX === "undefined" ) { alert( "No se pudo cargar la herramienta de Excel. Verifica tu conexión a Internet." ); return; } // ========================================= // CARGAR TODAS LAS ACTIVIDADES // ========================================= const { data: todasActividades, error: errorActividades } = await supabaseClient .from("actividades") .select( "id,curso,trimestre,categoria,nombre,orden,puntaje_maximo" ) .eq( "trimestre", 3 ) .order( "orden" ); if (errorActividades) { throw errorActividades; } // ========================================= // CARGAR TODAS LAS CALIFICACIONES // ========================================= const idsActividades = (todasActividades || []) .map( a => a.id ); let todasCalificaciones = []; if ( idsActividades.length > 0 ) { const { data, error } = await supabaseClient .from("calificaciones") .select( "actividad_id,estudiante_id,nota" ) .in( "actividad_id", idsActividades ); if (error) { throw error; } todasCalificaciones = data || []; } // ========================================= // MAPA DE CALIFICACIONES // ========================================= const mapaCalificaciones = {}; todasCalificaciones.forEach( cal => { const clave = cal.estudiante_id + "_" + cal.actividad_id; mapaCalificaciones[ clave ] = cal.nota; }); // ========================================= // CURSOS // ========================================= const cursos = [...new Set( estudiantes .map( e => e.curso ) .filter(Boolean) )].sort(); if (cursos.length === 0) { alert( "No existen estudiantes para exportar." ); return; } // ========================================= // CREAR LIBRO // ========================================= const libro = XLSX.utils.book_new(); // ========================================= // UNA HOJA POR CURSO // ========================================= cursos.forEach( curso => { const estudiantesCurso = estudiantes.filter( e => e.curso === curso ); const actividadesCurso = (todasActividades || []) .filter( a => a.curso === curso ); const saber = actividadesCurso.filter( a => a.categoria === "SABER" ); const hacer = actividadesCurso.filter( a => a.categoria === "HACER" ); const ser = actividadesCurso.filter( a => a.categoria === "SER" ); const auto = actividadesCurso.filter( a => a.categoria === "AUTOEVALUACION" ); // ====================================== // COLUMNAS // ====================================== const columnasSaber = ["PROMEDIO"] .concat( saber.map( a => a.nombre ) ); const columnasHacer = ["PROMEDIO"] .concat( hacer.map( a => a.nombre ) ); const columnasSer = ["PROMEDIO"] .concat( ser.map( a => a.nombre ) ); const columnasAuto = ["AUTOEVALUACIÓN"] .concat( auto.map( a => a.nombre ) ); const totalColumnas = 2 + columnasSer.length + columnasSaber.length + columnasHacer.length + columnasAuto.length + 1; // ====================================== // FILA 1 // ====================================== const fila1 = Array( totalColumnas ).fill(""); fila1[0] = "Nº"; fila1[1] = "APELLIDOS Y NOMBRES"; let posicion = 2; fila1[posicion] = "SER"; posicion += columnasSer.length; fila1[posicion] = "SABER"; posicion += columnasSaber.length; fila1[posicion] = "HACER"; posicion += columnasHacer.length; fila1[posicion] = "AUTOEVALUACIÓN"; posicion += columnasAuto.length; fila1[posicion] = "TOTAL"; // ====================================== // FILA 2 // ====================================== const fila2 = Array( totalColumnas ).fill(""); posicion = 2; fila2[posicion] = 10; posicion += columnasSer.length; fila2[posicion] = 45; posicion += columnasSaber.length; fila2[posicion] = 40; posicion += columnasHacer.length; fila2[posicion] = 5; // ====================================== // FILA 3 // ====================================== const fila3 = Array( totalColumnas ).fill(""); posicion = 2; columnasSer.forEach( nombre => { fila3[posicion] = nombre; posicion++; }); columnasSaber.forEach( nombre => { fila3[posicion] = nombre; posicion++; }); columnasHacer.forEach( nombre => { fila3[posicion] = nombre; posicion++; }); columnasAuto.forEach( nombre => { fila3[posicion] = nombre; posicion++; }); fila3[posicion] = "TOTAL"; // ====================================== // DATOS // ====================================== const datos = [ fila1, fila2, fila3 ]; estudiantesCurso.forEach( estudiante => { const fila = Array( totalColumnas ).fill(""); fila[0] = estudiante.num_lista ?? ""; fila[1] = estudiante.nombre ?? ""; let col = 2; // ================================== // SER // ================================== const notasSer = ser.map( actividad => obtenerNotaExcel( estudiante.id, actividad.id, mapaCalificaciones ) ); fila[col] = calcularPromedioPonderado( notasSer, ser ); col++; ser.forEach( actividad => { fila[col] = obtenerNotaExcel( estudiante.id, actividad.id, mapaCalificaciones ); col++; }); // ================================== // SABER // ================================== const notasSaber = saber.map( actividad => obtenerNotaExcel( estudiante.id, actividad.id, mapaCalificaciones ) ); fila[col] = calcularPromedioPonderado( notasSaber, saber ); col++; saber.forEach( actividad => { fila[col] = obtenerNotaExcel( estudiante.id, actividad.id, mapaCalificaciones ); col++; }); // ================================== // HACER // ================================== const notasHacer = hacer.map( actividad => obtenerNotaExcel( estudiante.id, actividad.id, mapaCalificaciones ) ); fila[col] = calcularPromedioPonderado( notasHacer, hacer ); col++; hacer.forEach( actividad => { fila[col] = obtenerNotaExcel( estudiante.id, actividad.id, mapaCalificaciones ); col++; }); // ================================== // AUTOEVALUACIÓN // ================================== const notasAuto = auto.map( actividad => obtenerNotaExcel( estudiante.id, actividad.id, mapaCalificaciones ) ); fila[col] = calcularPromedioPonderado( notasAuto, auto ); col++; auto.forEach( actividad => { fila[col] = obtenerNotaExcel( estudiante.id, actividad.id, mapaCalificaciones ); col++; }); // ================================== // TOTAL // ================================== fila[col] = estudiante.nota_3t ?? ""; datos.push( fila ); }); // ====================================== // CREAR HOJA // ====================================== const hoja = XLSX.utils.aoa_to_sheet( datos ); // ====================================== // COMBINAR ENCABEZADOS // ====================================== const merges = []; // Nº merges.push({ s: { r: 0, c: 0 }, e: { r: 2, c: 0 } }); // NOMBRE merges.push({ s: { r: 0, c: 1 }, e: { r: 2, c: 1 } }); let inicio = 2; // SER merges.push({ s: { r: 0, c: inicio }, e: { r: 0, c: inicio + columnasSer.length - 1 } }); inicio += columnasSer.length; // SABER merges.push({ s: { r: 0, c: inicio }, e: { r: 0, c: inicio + columnasSaber.length - 1 } }); inicio += columnasSaber.length; // HACER merges.push({ s: { r: 0, c: inicio }, e: { r: 0, c: inicio + columnasHacer.length - 1 } }); inicio += columnasHacer.length; // AUTO merges.push({ s: { r: 0, c: inicio }, e: { r: 0, c: inicio + columnasAuto.length - 1 } }); inicio += columnasAuto.length; // TOTAL merges.push({ s: { r: 0, c: inicio }, e: { r: 2, c: inicio } }); hoja["!merges"] = merges; // ====================================== // ANCHO // ====================================== hoja["!cols"] = Array( totalColumnas ) .fill(null) .map( (_, indice) => { if (indice === 0) { return { wch: 8 }; } if (indice === 1) { return { wch: 32 }; } return { wch: 18 }; }); // ====================================== // ALTURA // ====================================== hoja["!rows"] = [ { hpt: 25 }, { hpt: 22 }, { hpt: 40 } ]; // ====================================== // ESTILO // ====================================== const rango = XLSX.utils.decode_range( hoja["!ref"] ); for ( let fila = rango.s.r; fila <= rango.e.r; fila++ ) { for ( let columna = rango.s.c; columna <= rango.e.c; columna++ ) { const celda = hoja[ XLSX.utils.encode_cell({ r: fila, c: columna }) ]; if (!celda) { continue; } celda.s = { alignment: { horizontal: "center", vertical: "center", wrapText: true }, font: { name: "Arial", sz: 10 }, border: { top: { style: "thin" }, bottom: { style: "thin" }, left: { style: "thin" }, right: { style: "thin" } } }; } } // ====================================== // NOMBRE A LA IZQUIERDA // ====================================== for ( let fila = 3; fila <= rango.e.r; fila++ ) { const celda = hoja[ XLSX.utils.encode_cell({ r: fila, c: 1 }) ]; if (celda) { celda.s = celda.s || {}; celda.s.alignment = { horizontal: "left", vertical: "center" }; } } // ====================================== // NOMBRE DE HOJA // ====================================== let nombreHoja = String(curso) .replace( /[\\\/\?\*\[\]:]/g, "" ) .substring( 0, 31 ); if (!nombreHoja) { nombreHoja = "Curso"; } XLSX.utils.book_append_sheet( libro, hoja, nombreHoja ); }); // ========================================= // DESCARGAR EXCEL // ========================================= const fecha = new Date() .toISOString() .slice( 0, 10 ); XLSX.writeFile( libro, "Calificaciones_Tercer_Trimestre_" + fecha + ".xlsx" ); mostrarMensaje( "✓ Notas guardadas y Excel exportado correctamente." ); } catch (error) { console.error(error); mostrarMensaje( "❌ Error al exportar Excel: " + error.message, true ); } }
